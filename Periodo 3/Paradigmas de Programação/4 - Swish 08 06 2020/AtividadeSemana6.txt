% Regras para classes.
classe(X, mamifero) :- temPelo(X).
classe(X, peixe) :- viveNaAgua(X).
classe(X, ave) :- temPenas(X).
classe(X, inseto) :- poeOvos(X), not(viveNaAgua(X)), not(vertebrado(X)), not(temPenas(X)), not(temPelo(X)).
classe(X, reptil) :- poeOvos(X), vertebrado(X), not(viveNaAgua(X)), not(temPenas(X)), not(temPelo(X)).

% Imprime a classe de cada objeto numa lista.
classes([H|T], X) :- classe(H, X), T = [].
classes([H|T], X) :- classe(H, X) ; classes(T, X).
