CREATE OR REPLACE FUNCTION modificacao_neg_item_venda() RETURNS TRIGGER AS
$$
	BEGIN
		IF NEW.quantidade < 0 THEN
			RAISE EXCEPTION 'Coluna "quantidade" não permite valores negativos (Quantidade inserida: %).', NEW.quantidade;
			RETURN NULL;
		END IF;
		
		RETURN NEW;
	END;
$$
LANGUAGE plpgsql;
